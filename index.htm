      <div class="photo-icon">📸</div>

      <input class="hidden-file-input" type="file" id="fileInput" accept="image/jpeg,image/png,image/webp,image/gif,image/heic,image/heif" multiple>

      <button class="action-button" id="actionBtn" onclick="handleButtonClick()">Select Photos</button>

      <div id="status"></div>

    <div class="footer">♥ Thank you for sharing your memories!</div>

  <script>

    const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbz_lnK43BK_s_esE1QWEMdkHwiGWmqbiuoC1r6ZBkbVZJgAvHpzBmG0SlDYFX461JVAnw/exec";

    const fileInput = document.getElementById("fileInput");

    const actionBtn = document.getElementById("actionBtn");

    const status = document.getElementById("status");


    fileInput.addEventListener("change", function() {

      if (fileInput.files.length > 0) {

        const count = fileInput.files.length;

        actionBtn.innerText = `Upload ${count} Photo${count > 1 ? 's' : ''}`;

        actionBtn.classList.add("ready");

        status.className = "info";

        status.innerText = `${count} photo${count > 1 ? 's' : ''} ready to upload.`;

      } else {

        resetButton();

      }

    });


    function handleButtonClick() {

      if (fileInput.files.length === 0) {

        fileInput.click();

      } else {

        uploadPhotos();

      }

    }


    function resetButton() {

      fileInput.value = "";

      actionBtn.innerText = "Select Photos";

      actionBtn.className = "action-button";

      actionBtn.disabled = false;

    }


    function uploadPhotos() {

      actionBtn.disabled = true;

      actionBtn.innerText = "Uploading...";

      status.className = "";

      status.innerText = "Please wait while your photos are uploaded...";

      uploadNextFile(Array.from(fileInput.files), 0);

    }


    function uploadNextFile(files, index) {

      if (index >= files.length) {

        status.className = "success";

        status.innerText = "All photos uploaded successfully! 🎉";

        resetButton();

        return;

      }


      const file = files[index];

      status.className = "";

      status.innerText = `Uploading photo ${index + 1} of ${files.length}...`;


      if (!file.type.startsWith("image/")) {

        status.className = "error";

        status.innerText = "Upload stopped. Only photos are allowed.";

        resetButton();

        return;

      }


      const reader = new FileReader();

      reader.onload = function(e) {

        const base64Data = e.target.result.split(',')[1];

        const payload = { name: file.name, mimeType: file.type, base64: base64Data };


        fetch(SCRIPT_URL, { method: "POST", body: JSON.stringify(payload) })

        .then(res => res.json())

        .then(result => {

          if (!result.success) {

            status.className = "error";

            status.innerText = "Error: " + result.message;

            resetButton();

            return;

          }

          uploadNextFile(files, index + 1);

        })

        .catch(() => {

          status.className = "error";

          status.innerText = "Upload failed. Please try again.";

          resetButton();

        });

      };

      reader.readAsDataURL(file);

    }

  </script>
